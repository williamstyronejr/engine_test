cmake_minimum_required(VERSION 3.21)
project(EngineTest VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This engine supports Linux only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ENGINE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(ENGINE_BUILD_PLATFORM "Build Linux/OpenGL targets (OFF for CPU-only tests)" ON)
add_library(engine_options INTERFACE)
target_compile_options(engine_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(ENGINE_SANITIZERS)
  target_compile_options(engine_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(engine_options INTERFACE -fsanitize=address,undefined)
endif()
add_library(engine_core STATIC engine/src/assets.cpp engine/src/scene.cpp engine/src/tilemap.cpp engine/src/animation.cpp engine/src/sound.cpp engine/src/ui.cpp engine/src/persistence.cpp engine/src/collision_world.cpp)
target_include_directories(engine_core PUBLIC engine/include)
target_link_libraries(engine_core PUBLIC engine_options)
add_executable(make_assets tools/make_assets.cpp)
target_link_libraries(make_assets PRIVATE engine_core)
include(CTest)
